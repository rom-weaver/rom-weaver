	r2 = (a->format->read_header)(a, entry);
	if (r2 == ARCHIVE_EOF) {
		struct archive_read_filter *filter;

		/* Each gzip filter MUST be drained directly: downstream decoders
		 * can finish before upstream members and trailers are validated. */
		for (filter = a->filter; filter != NULL; filter = filter->upstream) {
			if (filter->code != ARCHIVE_FILTER_GZIP)
				continue;
			for (;;) {
				ssize_t available;

				if (__archive_read_filter_ahead(filter, 1,
				    &available) == NULL) {
					if (available < 0)
						r2 = ARCHIVE_FATAL;
					break;
				}
				if (__archive_read_filter_consume(filter,
				    available) < 0) {
					r2 = ARCHIVE_FATAL;
					break;
				}
			}
			if (r2 == ARCHIVE_FATAL)
				break;
		}
